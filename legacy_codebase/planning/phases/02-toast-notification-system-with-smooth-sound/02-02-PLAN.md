Create `resources/js/Components/ToastManager.svelte` using Svelte 5 runes:

```svelte
<script>
  import { toast } from 'svelte-french-toast';
  import '../lib/toast.css';

  // Sound generation using Web Audio API (D-04: subtle chime)
  let audioContext = $state(null);

  function getAudioContext() {
    if (typeof window !== 'undefined' && !audioContext) {
      audioContext = new (window.AudioContext || window.webkitAudioContext)();
    }
    return audioContext;
  }

  export function playSound() {
    const ctx = getAudioContext();
    if (!ctx) return;

    // Resume context if suspended (autoplay policy)
    if (ctx.state === 'suspended') {
      ctx.resume();
    }

    const oscillator = ctx.createOscillator();
    const gainNode = ctx.createGain();

    oscillator.connect(gainNode);
    gainNode.connect(ctx.destination);

    // Pleasant chime sweep (800Hz -> 400Hz)
    oscillator.frequency.setValueAtTime(800, ctx.currentTime);
    oscillator.frequency.exponentialRampToValueAtTime(400, ctx.currentTime + 0.3);

    // Quick fade out
    gainNode.gain.setValueAtTime(0.2, ctx.currentTime);
    gainNode.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.5);

    oscillator.start(ctx.currentTime);
    oscillator.stop(ctx.currentTime + 0.5);
  }

  // Convenience methods matching svelte-french-toast API
  export function success(message, options = {}) {
    playSound();
    return toast.success(message, { duration: 4000, ...options });
  }

  export function error(message, options = {}) {
    playSound();
    return toast.error(message, { duration: 4000, ...options });
  }

  export function message(message, options = {}) {
    playSound();
    return toast(message, { duration: 4000, ...options });
  }

  // Static toast without sound (for non-event notifications)
  export function silent(message, options = {}) {
    return toast(message, { duration: 4000, ...options });
  }
</script>

```
